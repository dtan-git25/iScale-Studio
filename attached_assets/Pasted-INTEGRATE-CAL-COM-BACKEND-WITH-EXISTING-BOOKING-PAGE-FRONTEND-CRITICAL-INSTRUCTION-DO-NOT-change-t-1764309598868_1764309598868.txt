STEP 3: CREATE BACKEND API ROUTES
═══════════════════════════════════════════════════════════

Create file: `/server/routes/cal.js` (or appropriate backend location)
```javascriptconst express = require('express');
const router = express.Router();
const axios = require('axios');const CAL_API_KEY = process.env.CALCOM_API_KEY;
const CAL_EVENT_TYPE_ID = process.env.CALCOM_EVENT_TYPE_ID;
const CAL_API_URL = 'https://api.cal.com/v1';// Middleware to check API key
const checkApiKey = (req, res, next) => {
if (!CAL_API_KEY) {
return res.status(500).json({ error: 'Cal.com API key not configured' });
}
next();
};// GET AVAILABLE SLOTS FOR A DATE
router.get('/availability', checkApiKey, async (req, res) => {
const { date } = req.query;if (!date) {
return res.status(400).json({ error: 'Date parameter required' });
}try {
// Format: YYYY-MM-DD to ISO datetime
const startTime = ${date}T00:00:00Z;
const endTime = ${date}T23:59:59Z;const response = await axios.get(
  `${CAL_API_URL}/slots`,
  {
    params: {
      eventTypeId: CAL_EVENT_TYPE_ID,
      startTime: startTime,
      endTime: endTime
    },
    headers: {
      'Authorization': `Bearer ${CAL_API_KEY}`
    }
  }
);// Transform Cal.com slots to frontend format
const slots = response.data.slots.map(slot => {
  const slotTime = new Date(slot.time);
  return {
    time: slotTime.toLocaleTimeString('en-US', { 
      hour: 'numeric', 
      minute: '2-digit',
      hour12: true 
    }),
    available: true,
    rawTime: slot.time
  };
});res.json({ success: true, slots });
} catch (error) {
console.error('Error fetching availability:', error.response?.data || error.message);
res.status(500).json({
success: false,
error: 'Failed to fetch availability',
details: error.response?.data
});
}
});// CREATE BOOKING
router.post('/book', checkApiKey, async (req, res) => {
const { selectedSlot, name, email, phone, company, service, message } = req.body;// Validation
if (!selectedSlot || !name || !email) {
return res.status(400).json({
success: false,
error: 'Missing required fields: selectedSlot, name, email'
});
}try {
const bookingData = {
eventTypeId: parseInt(CAL_EVENT_TYPE_ID),
start: selectedSlot, // Should be ISO datetime from frontend
responses: {
name: name,
email: email,
guests: [],
location: { optionValue: '', value: 'integrations:zoom' }
},
metadata: {
phone: phone || '',
company: company || '',
service: service || 'AI Web & App Development',
notes: message || ''
},
timeZone: 'Asia/Manila',
language: 'en'
};const response = await axios.post(
  `${CAL_API_URL}/bookings`,
  bookingData,
  {
    headers: {
      'Authorization': `Bearer ${CAL_API_KEY}`,
      'Content-Type': 'application/json',
      'cal-api-version': '2024-08-13'
    }
  }
);res.json({ 
  success: true, 
  booking: response.data,
  message: 'Booking confirmed! Check your email for calendar invite and Zoom link.' 
});
} catch (error) {
console.error('Booking error:', error.response?.data || error.message);
res.status(500).json({
success: false,
error: 'Failed to create booking',
details: error.response?.data
});
}
});// GET UPCOMING BOOKINGS (for dashboard later)
router.get('/bookings', checkApiKey, async (req, res) => {
try {
const response = await axios.get(
${CAL_API_URL}/bookings,
{
headers: {
'Authorization': Bearer ${CAL_API_KEY}
}
}
);res.json({ success: true, bookings: response.data });
} catch (error) {
console.error('Error fetching bookings:', error.response?.data || error.message);
res.status(500).json({
success: false,
error: 'Failed to fetch bookings'
});
}
});module.exports = router;
